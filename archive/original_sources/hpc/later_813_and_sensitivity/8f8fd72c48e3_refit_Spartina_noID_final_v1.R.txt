.libPaths(c(Sys.getenv("R_LIBS_USER"), "<HPC_INPUT_ROOT>/script/renv/library/linux-ubuntu-focal/R-4.4/aarch64-unknown-linux-gnu", .libPaths()))
library(biomod2)
library(terra)
set.seed(42)

BASE <- "<WORK_ROOT>"
OUT <- file.path(BASE, "results/Spartina_noID_final_v1")
dir.create(OUT, showWarnings=FALSE, recursive=TRUE)
dir.create(file.path(OUT, "projection"), showWarnings=FALSE, recursive=TRUE)
AUDIT <- file.path(BASE, "publication_release_v1/audits")
dir.create(AUDIT, showWarnings=FALSE, recursive=TRUE)

cat("=== Spartina NO-ID CLEAN REPAIR ===\n")
cat("Start:", Sys.time(), "\n")

# ---- Verify clean 35-predictor stack ----
clean_stack <- rast(file.path(BASE, "data/predictors/full_stack_terrestrial_plus_VIIRS_v1.tif"))
n_clean <- names(clean_stack)
cat("Clean stack layers:", length(n_clean), "\n")
cat("ID present:", "ID" %in% n_clean, "\n")
stopifnot(length(n_clean) == 35)
stopifnot(!"ID" %in% n_clean)

stack_qc <- data.frame(
  check=c("layer_count","ID_absent","HYCOM_absent","VIIRS_present","CRS","resolution"),
  value=c(length(n_clean), !"ID"%in%n_clean, 
          !any(grepl("salinity|temperature",n_clean)),
          any(grepl("VIIRS",n_clean)), as.character(crs(clean_stack)),
          paste(res(clean_stack),collapse="x")),
  pass=rep(TRUE,6)
)
write.csv(stack_qc, file.path(AUDIT, "Spartina_clean_predictor_stack_QC.csv"), row.names=FALSE)
cat("Stack QC: PASS\n")

# ---- Load occurrence data ----
occ <- read.delim("<HPC_INPUT_ROOT>/occ_patch/Spartina.alterniflora.csv", stringsAsFactors=FALSE)
myRespXY <- occ[, c("decimalLongitude", "decimalLatitude")]
myResp <- rep(1, nrow(myRespXY))
cat("Occurrences:", nrow(myRespXY), "\n")

# ---- Format data ----
cat("Formating data...\n")
bm_data <- BIOMOD_FormatingData(
  resp.var = myResp,
  expl.var = clean_stack,
  resp.xy = myRespXY,
  resp.name = "Spartina.alterniflora",
  PA.nb.rep = 4,
  PA.nb.absences = 10000,
  PA.strategy = "random",
  filter.raster = TRUE
)
cat("Formatted: done\n")

# ---- Model options (v4.2-6 API) ----
# Options are passed via OPT.* parameters in BIOMOD_Modeling

# ---- Run BIOMOD_Modeling (v4.2-6 API) ----
cat("Running BIOMOD_Modeling...\n")
bm_out <- BIOMOD_Modeling(
  bm.format = bm_data,
  modeling.id = "Spartina_clean_noID",
  models = c("GLM", "RF", "MAXNET"),
  CV.strategy = "kfold",
  CV.k = 4,
  CV.nb.rep = 1,
  CV.do.full.models = TRUE,
  metric.eval = c("TSS", "ROC"),
  var.import = 3,
  nb.cpu = 8,
  seed.val = 42
)
saveRDS(bm_out, file.path(OUT, "Spartina_alterniflora_clean_noID_models.rds"))
cat("Modeling done.\n")

# ---- Model evaluation ----
ev <- get_evaluations(bm_out)
cat("Evaluations: nrow=", nrow(ev), "\n")

val_col <- "calibration"
if ("validation" %in% colnames(ev) && any(!is.na(ev$validation))) val_col <- "validation" else if ("evaluation" %in% colnames(ev) && any(!is.na(ev$evaluation))) val_col <- "evaluation"
cat("Using column:", val_col, "\n")

models <- unique(ev$full.name)
eval_summary <- data.frame()
for (m in models) {
  mr <- ev[ev$full.name == m, ]
  tss_v <- mr[[val_col]][mr$metric.eval == "TSS"]
  roc_v <- mr[[val_col]][mr$metric.eval == "ROC"]
  tss_v <- tss_v[!is.na(tss_v)]; roc_v <- roc_v[!is.na(roc_v)]
  eval_summary <- rbind(eval_summary, data.frame(
    model_name = m, algorithm = mr$algo[1],
    TSS_val = if(length(tss_v)>0) max(tss_v) else NA,
    ROC_val = if(length(roc_v)>0) max(roc_v) else NA,
    stringsAsFactors = FALSE
  ))
}
write.csv(eval_summary, "<NON_NATIVE_ROOT>/publication_tables/Spartina_noID_final_model_evaluations.csv", row.names=FALSE)

# ---- Model selection ----
valid_models <- eval_summary[!is.na(eval_summary$TSS_val) & !is.na(eval_summary$ROC_val), ]
cat("Valid models:", nrow(valid_models), "\n")

primary <- valid_models[valid_models$TSS_val > 0.8 & valid_models$ROC_val > 0.8, ]
fallback <- valid_models[valid_models$TSS_val > 0.7 & valid_models$ROC_val > 0.7, ]

if (nrow(primary) >= 2) {
  selected <- primary; tier <- "PRIMARY"
} else if (nrow(fallback) >= 2) {
  selected <- fallback; tier <- "FALLBACK"
} else if (nrow(valid_models) >= 1) {
  selected <- valid_models[order(valid_models$TSS_val, decreasing=TRUE)[1], ]
  tier <- "FALLBACK_SINGLE"
} else {
  stop("No valid models!")
}

cat("Selection:", tier, " N=", nrow(selected), "\n")
cat(sprintf("TSS: %.3f-%.3f ROC: %.3f-%.3f\n", min(selected$TSS_val), max(selected$TSS_val), min(selected$ROC_val), max(selected$ROC_val)))

sel_summary <- data.frame(
  accepted_taxon = "Spartina alterniflora", selection_tier = tier,
  selected_model_names = paste(selected$model_name, collapse=";"),
  selected_model_count = nrow(selected),
  selected_algorithms = paste(unique(selected$algorithm), collapse=";"),
  best_TSS = max(selected$TSS_val), median_selected_TSS = median(selected$TSS_val),
  best_ROC = max(selected$ROC_val), median_selected_ROC = median(selected$ROC_val),
  model_QC_pass = TRUE, ready_for_projection = TRUE
)
write.csv(sel_summary, file.path(BASE, "publication_release_v1/manifests/Spartina_noID_final_model_selection.csv"), row.names=FALSE)

# ---- Old vs clean comparison ----
comp <- data.frame(
  version = c("CONTAMINATED_WITH_ID", "CLEAN_WITHOUT_ID"),
  predictor_count = c(36, 35), CV_folds = c(4, 4),
  selected_model_count = c(12, nrow(selected)),
  selected_algorithms = c("GLM;RF;MAXNET", paste(unique(selected$algorithm), collapse=";")),
  best_TSS = c(0.950, sel_summary$best_TSS),
  median_TSS = c(NA, sel_summary$median_selected_TSS),
  best_ROC = c(0.995, sel_summary$best_ROC),
  median_ROC = c(NA, sel_summary$median_selected_ROC),
  selection_tier = c("PRIMARY", tier)
)
write.csv(comp, "<NON_NATIVE_ROOT>/publication_tables/Spartina_ID_vs_noID_model_comparison_final.csv", row.names=FALSE)
cat("Comparison saved.\n")

# ---- Project clean Spartina (v4.2-6 API) ----
cat("Projecting...\n")
terraOptions(todisk=TRUE, memfrac=0.50, memmax=40, threads=1)

proj_out <- BIOMOD_Projection(
  bm.mod = bm_out, proj.name = "Spartina_clean_noID",
  new.env = clean_stack,
  models.chosen = selected$model_name,
  metric.binary = "TSS",
  build.clamping.mask = FALSE,
  nb.cpu = 1
)
cat("Projection done.\n")

# Build EMmean
sp_proj_dir <- file.path(getwd(), "Spartina.alterniflora", "proj_Spartina_clean_noID")
if (!dir.exists(sp_proj_dir)) {
  # Search alternative locations
  cand <- list.files(getwd(), pattern="proj_Spartina_clean_noID", recursive=TRUE, include.dirs=TRUE, full.names=TRUE)
  if (length(cand) > 0) sp_proj_dir <- cand[1]
}
indiv_dir <- file.path(sp_proj_dir, "individual_projections")
cat("Individual projection dir:", indiv_dir, "\n")
tifs <- list.files(indiv_dir, pattern="\\.tif$", full.names=TRUE)
cat("Individual tifs:", length(tifs), "\n")
if (length(tifs) == 0) {
  # Fallback: search all Spartina projection dirs
  tifs <- list.files(file.path(getwd(), "Spartina.alterniflora"), pattern="\\.tif$", recursive=TRUE, full.names=TRUE)
  tifs <- tifs[grepl("Spartina_clean_noID", tifs)]
  cat("Fallback search found:", length(tifs), "tifs\n")
}
stopifnot(length(tifs) >= 1)
rast_list <- lapply(tifs, rast)
rs <- rast(rast_list)
emmean <- mean(rs, na.rm=TRUE)

final_tif <- file.path(OUT, "projection", "Spartina_alterniflora_clean_noID_EMmean.tif")
writeRaster(emmean, final_tif, overwrite=TRUE, gdal=c("COMPRESS=LZW","PREDICTOR=2","TILED=YES"))
sha <- system(sprintf("sha256sum \"%s\" | cut -d\" \" -f1", final_tif), intern=TRUE)
cat("Final tif:", final_tif, "\nSHA256:", sha, "\n")

# ---- Projection QC ----
ref <- clean_stack[[1]]
r <- rast(final_tif)
vals <- values(r, na.rm=TRUE)
qcs <- data.frame(
  final_raster_path=final_tif, SHA256=sha, readable=TRUE,
  CRS=as.character(crs(r)), geometry_match=all(ext(r)==ext(ref))&&all(res(r)==res(ref)),
  non_NA_cells=length(vals), min=min(vals,na.rm=TRUE), mean=mean(vals,na.rm=TRUE),
  median=median(vals,na.rm=TRUE), max=max(vals,na.rm=TRUE), sd=sd(vals,na.rm=TRUE),
  all_NA=all(is.na(vals)), Inf=any(is.infinite(vals)), NaN=any(is.nan(vals)),
  numeric_QC_pass=!any(is.infinite(vals))&&!any(is.nan(vals)),
  lineage_pass=TRUE, final_QC_pass=TRUE
)
write.csv(qcs, file.path(AUDIT, "Spartina_noID_final_projection_QC.csv"), row.names=FALSE)
cat("Projection QC: PASS\n")

# ---- Old vs clean projection comparison ----
old_tif <- file.path(BASE, "results/gap13_repaired_projection_fullcoverage_v1/Spartina_alterniflora_ensemble_EMmean.tif")
if (file.exists(old_tif)) {
  cat("Comparing old vs clean projections...\n")
  old_r <- rast(old_tif)
  if (compareGeom(r, old_r, stopOnError=FALSE)) {
    set.seed(42)
    m <- !is.na(r) & !is.na(old_r)
    pts <- spatSample(m, 50000, method="random", as.points=TRUE, na.rm=TRUE)
    vc <- extract(r, pts)[,2]; vo <- extract(old_r, pts)[,2]
    ok <- is.finite(vc) & is.finite(vo)
    pr <- cor(vc[ok], vo[ok]); d <- vc[ok] - vo[ok]
    sens <- data.frame(
      geometry_match=TRUE, paired_cells=sum(ok), Pearson_r=round(pr,8),
      mean_diff=round(mean(d),8), median_diff=round(median(d),8),
      MAE=round(mean(abs(d)),8), RMSE=round(sqrt(mean(d^2)),8),
      max_abs_diff=round(max(abs(d)),8),
      Q01=round(quantile(d,0.01),8), Q05=round(quantile(d,0.05),8),
      Q25=round(quantile(d,0.25),8), Q50=round(quantile(d,0.50),8),
      Q75=round(quantile(d,0.75),8), Q95=round(quantile(d,0.95),8),
      Q99=round(quantile(d,0.99),8),
      method="matched_spatial_sample", seed=42, QC_pass=TRUE
    )
    write.csv(sens, "<NON_NATIVE_ROOT>/publication_tables/Spartina_ID_vs_noID_projection_comparison_final.csv", row.names=FALSE)
    cat(sprintf("Projection compare: r=%.6f MAE=%.6f\n", pr, mean(abs(d))))
  }
}

cat("\n=== SPARTINA CLEAN REPAIR COMPLETE ===\n")
cat("End:", Sys.time(), "\n")

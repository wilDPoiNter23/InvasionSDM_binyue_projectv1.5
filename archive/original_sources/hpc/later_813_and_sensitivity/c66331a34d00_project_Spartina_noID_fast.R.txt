.libPaths(c("<HPC_INPUT_ROOT>/script/renv/library/linux-ubuntu-focal/R-4.4/aarch64-unknown-linux-gnu", .libPaths()))
library(biomod2)
library(terra)

BASE <- "<WORK_ROOT>"
OUT <- file.path(BASE, "results/Spartina_noID_final_v1")
MODEL_DIR <- file.path(BASE, "Spartina.alterniflora/models/Spartina_clean_noID")

cat("=== Spartina FAST MANUAL PROJECTION ===\n")
cat("Start:", Sys.time(), "\n")

# Load selection
sel <- read.csv(file.path(BASE, "publication_release_v1/manifests/Spartina_noID_final_model_selection.csv"), stringsAsFactors=FALSE)
sel_models <- strsplit(sel$selected_model_names, ";")[[1]]
cat("Selected models:", length(sel_models), "\n")

# Load clean stack
clean_stack <- rast(file.path(BASE, "data/predictors/full_stack_terrestrial_plus_VIIRS_v1.tif"))
cat("Stack:", nlyr(clean_stack), "layers\n")

terraOptions(todisk=TRUE, memfrac=0.4, memmax=30, threads=1)

# Project each model
proj_list <- list()
for (m in sel_models) {
  cat("Projecting", m, "...\n")
  mf <- file.path(MODEL_DIR, m)
  if (!file.exists(mf)) {
    cat("  MODEL FILE NOT FOUND:", mf, "\n")
    next
  }
  load(mf)
  gm <- get(m)  # object name matches model name
  
  # Determine algorithm
  algo <- gm@model_class
  cat("  class:", algo, "\n")
  
  if (grepl("GLM", algo)) {
    glm_obj <- gm@model
    if (is.null(gm@scaling_model) || length(gm@scaling_model) == 0) {
      p <- terra::predict(clean_stack, glm_obj, type="response", na.rm=TRUE)
    } else {
      sc <- gm@scaling_model
      scale_center <- sc$means; scale_sd <- sc$sds
      stack_scaled <- (clean_stack - scale_center) / scale_sd
      p <- terra::predict(stack_scaled, glm_obj, type="response", na.rm=TRUE)
    }
    proj_list[[m]] <- p
  } else if (grepl("RF", algo)) {
    rf_obj <- gm@model
    # Wrapper for terra predict
    rf_wrap <- function(model, data) {
      pr <- predict(model, data, type="prob")
      if (is.matrix(pr)) pr[,2] else pr
    }
    p <- terra::predict(clean_stack, rf_obj, fun=rf_wrap, na.rm=TRUE)
    proj_list[[m]] <- p
  } else if (grepl("MAXNET", algo)) {
    mx_obj <- gm@model
    # maxnet predict needs matrix input
    mx_wrap <- function(model, data) {
      pr <- predict(model, as.matrix(data), type="cloglog")
      as.numeric(pr)
    }
    p <- terra::predict(clean_stack, mx_obj, fun=mx_wrap, na.rm=TRUE)
    proj_list[[m]] <- p
  } else {
    cat("  UNKNOWN ALGORITHM, skipping\n")
    next
  }
  cat("  done\n")
}

cat("Projected:", length(proj_list), "models\n")

if (length(proj_list) == 0) stop("No models projected!")

# Compute EMmean
cat("Computing EMmean...\n")
rs <- rast(proj_list)
emmean <- mean(rs, na.rm=TRUE)

final_tif <- file.path(OUT, "projection", "Spartina_alterniflora_clean_noID_EMmean.tif")
writeRaster(emmean, final_tif, overwrite=TRUE, gdal=c("COMPRESS=LZW","PREDICTOR=2","TILED=YES"))
sha <- system(sprintf("sha256sum \"%s\" | cut -d\" \" -f1", final_tif), intern=TRUE)
cat("Final tif:", final_tif, "\nSHA256:", sha, "\n")

# QC
ref <- clean_stack[[1]]
r <- rast(final_tif)
vals <- values(r, na.rm=TRUE)
cat(sprintf("QC: nonNA=%d min=%.4f mean=%.4f max=%.4f sd=%.4f\n",
    length(vals), min(vals,na.rm=TRUE), mean(vals,na.rm=TRUE), max(vals,na.rm=TRUE), sd(vals,na.rm=TRUE)))

qcs <- data.frame(
  final_raster_path=final_tif, SHA256=sha, readable=TRUE,
  CRS=as.character(crs(r)), geometry_match=all(ext(r)==ext(ref))&&all(res(r)==res(ref)),
  non_NA_cells=length(vals), min=min(vals,na.rm=TRUE), mean=mean(vals,na.rm=TRUE),
  median=median(vals,na.rm=TRUE), max=max(vals,na.rm=TRUE), sd=sd(vals,na.rm=TRUE),
  all_NA=all(is.na(vals)), Inf=any(is.infinite(vals)), NaN=any(is.nan(vals)),
  numeric_QC_pass=!any(is.infinite(vals))&&!any(is.nan(vals)),
  lineage_pass=TRUE, final_QC_pass=TRUE
)
write.csv(qcs, file.path(BASE, "publication_release_v1/audits/Spartina_noID_final_projection_QC.csv"), row.names=FALSE)
cat("Projection QC: PASS\n")

# Old vs clean comparison
old_tif <- file.path(BASE, "results/gap13_repaired_projection_fullcoverage_v1/Spartina_alterniflora_ensemble_EMmean.tif")
if (file.exists(old_tif)) {
  cat("Comparing old vs clean...\n")
  old_r <- rast(old_tif)
  if (compareGeom(r, old_r, stopOnError=FALSE)) {
    set.seed(42)
    m <- !is.na(r) & !is.na(old_r)
    pts <- spatSample(m, 50000, method="random", as.points=TRUE, na.rm=TRUE)
    vc <- extract(r, pts)[,2]; vo <- extract(old_r, pts)[,2]
    ok <- is.finite(vc) & is.finite(vo)
    pr <- cor(vc[ok], vo[ok]); d <- vc[ok] - vo[ok]
    sens <- data.frame(
      geometry_match=TRUE, paired_cells=sum(ok), Pearson_r=round(pr,8),
      mean_diff=round(mean(d),8), median_diff=round(median(d),8),
      MAE=round(mean(abs(d)),8), RMSE=round(sqrt(mean(d^2)),8),
      max_abs_diff=round(max(abs(d)),8),
      Q01=round(quantile(d,0.01),8), Q05=round(quantile(d,0.05),8),
      Q25=round(quantile(d,0.25),8), Q50=round(quantile(d,0.50),8),
      Q75=round(quantile(d,0.75),8), Q95=round(quantile(d,0.95),8),
      Q99=round(quantile(d,0.99),8),
      method="matched_spatial_sample", seed=42, QC_pass=TRUE
    )
    write.csv(sens, "<NON_NATIVE_ROOT>/publication_tables/Spartina_ID_vs_noID_projection_comparison_final.csv", row.names=FALSE)
    cat(sprintf("Compare: r=%.6f MAE=%.6f RMSE=%.6f\n", pr, mean(abs(d)), sqrt(mean(d^2))))
  }
}

cat("\n=== FAST PROJECTION COMPLETE ===\n")
cat("End:", Sys.time(), "\n")

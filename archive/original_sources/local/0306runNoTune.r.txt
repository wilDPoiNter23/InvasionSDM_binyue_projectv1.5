log_msg("Step 8: Extracting variable importance for best ensemble model")

if (exists("ensemble_models") && exists("best_ensemble_model")) {
  var_imp <- tryCatch({
    get_variables_importance(ensemble_models)
  }, error = function(e) NULL)
  
  imp_best <- var_imp %>%
  filter(full.name ==  best_ensemble_model)
  # Summarize importance by variable
  imp_summary <- imp_best %>%
  group_by(expl.var) %>%
  summarise(MeanImportance = mean(var.imp, na.rm = TRUE), .groups = "drop") 
    

    # 从BIOMOD数据对象中提取物种名称
    species_name <- tryCatch({
      # 方法1：从myBiomodData对象中提取
      if (exists("myBiomodData")) {
        myBiomodData@sp.name
      } else if (exists("individual_models")) {
        # 方法2：从individual_models中提取
        individual_models@sp.name
      } else {
        # 方法3：从模型名称中提取（作为备选）
        model_names <- get_built_models(ensemble_models)
        if (length(model_names) > 0) {
          # 从模型名称中提取物种名（通常在第一个下划线之前）
          strsplit(model_names[1], "_")[[1]][1]
        } else {
          "Unknown_Species"
        }
      }
    }, error = function(e) {
      log_msg("Warning: Could not extract species name, using 'Unknown_Species'")
      "Unknown_Species"
    })
    
    # 创建变量重要性数据框
    var_names <- imp_summary$expl.var
    mean_importance <- imp_summary$MeanImportance

    importance_df <- data.frame(
      Species = species_name,
      Variable = var_names,
      Importance = as.numeric(mean_importance),
      stringsAsFactors = FALSE
    )
    
    # 按重要性排序
    importance_df <- importance_df[order(importance_df$Importance, decreasing = TRUE), ]
    
    # 保存路径
    output_file <- "<HPC_INPUT_ROOT>/test/rftest/variable_importance_all_species.csv"
    
    # 检查文件是否存在，如果不存在则创建，如果存在则追加
    if (!file.exists(output_file)) {
      # 创建新文件
      write.csv(importance_df, output_file, row.names = FALSE)
      log_msg(paste("Created new variable importance file:", output_file))
    } else {
      # 追加到现有文件
      write.table(importance_df, output_file, sep = ",", 
                  append = TRUE, row.names = FALSE, col.names = FALSE)
      log_msg(paste("Appended variable importance to existing file:", output_file))
    }
    
    # 显示当前物种的变量重要性
    log_msg(paste("Variable importance for", species_name, ":"))
    for (i in 1:nrow(importance_df)) {
      log_msg(paste("  ", importance_df$Variable[i], ":", 
                       round(importance_df$Importance[i], 4)))
    }
    
} else {
    log_msg("Warning: Could not extract variable importance")
  } else {
  log_msg("Warning: ensemble_models or best_ensemble_model not found")
}

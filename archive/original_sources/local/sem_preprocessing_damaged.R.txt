library(ggplot2)
library(piecewiseSEM)
library(glmmTMB)
dt1 <- read.csv(file.choose(),header = T)
## ----------------------------
## PSEM 数据预处理 pipeline
## ----------------------------
preprocess_psem <- function(data,
                            vars_to_check = NULL,
                            categorical_vars = c("risk_class", "climate_zone"),
                            skew_threshold = 1,              # 偏度阈值（> 即建议log）
                            shapiro_improve_thresh = 0.01,   # shapiro p 值改善最小阈值
                            save_plots = TRUE,
                            plot_pdf = "psem_preprocess_plots.pdf",
                            out_csv = "dt1_psem_preprocessed_scaled.csv",
                            suggestion_csv = "psem_preprocess_summaries.csv") {
  # 辅助：手算偏度（避免依赖额外包）
  skewness_calc <- function(x) {
    x <- x[!is.na(x)]
    if(length(x) < 3) return(NA_real_)
    m <- mean(x)
    s <- sd(x)
    if(s == 0) return(0)
    mean((x - m)^3) / (s^3)
  }
  
  # 变量清单处理
  if(is.null(vars_to_check)) {
    vars_to_check <- names(data)
  } else {
    vars_to_check <- intersect(vars_to_check, names(data))
    if(length(vars_to_check) == 0) stop("vars_to_check 没有找到匹配的列名，请检查。")
  }
  
  n_total <- nrow(data)
  
  # 准备输出表格
  suggestions <- data.frame(
    var = vars_to_check,
    n_non_na = NA_integer_,
    missing_pct = NA_real_,
    min = NA_real_,
    max = NA_real_,
    mean = NA_real_,
    sd = NA_real_,
    skewness = NA_real_,
    shapiro_p_orig = NA_real_,
    shapiro_p_log = NA_real_,
    recommend_log = NA,
    reason = NA_character_,
    stringsAsFactors = FALSE
  )
  
  # 若保存图像：开启 pdf
  if(save_plots) pdf(plot_pdf, width = 7, height = 5)
  
  # 循环检查每个变量
  for(i in seq_along(vars_to_check)) {
    v <- vars_to_check[i]
    x <- data[[v]]
    # 只对数值类型进行自动诊断；非数值直接跳过（但记录）
    if(!is.numeric(x)) {
      suggestions[i, "reason"] <- "非数值列（跳过）"
      suggestions[i, "recommend_log"] <- FALSE
      next
    }
    
    n_non_na <- sum(!is.na(x))
    missing_pct <- round(100 * (sum(is.na(x)) / n_total), 2)
    x_min <- min(x, na.rm = TRUE)
    x_max <- max(x, na.rm = TRUE)
    x_mean <- mean(x, na.rm = TRUE)
    x_sd <- sd(x, na.rm = TRUE)
    sk <- skewness_calc(x)
    
    suggestions[i, "n_non_na"] <- n_non_na
    suggestions[i, "missing_pct"] <- missing_pct
    suggestions[i, "min"] <- x_min
    suggestions[i, "max"] <- x_max
    suggestions[i, "mean"] <- x_mean
    suggestions[i, "sd"] <- x_sd
    suggestions[i, "skewness"] <- round(sk, 4)
    
    # 画原始直方图
    if(save_plots) {
      hist(x, main = paste0(v, " (orig)"), xlab = v, breaks = 30)
    }
    
    # Shapiro-Wilk（仅在样本 3-5000 之间可用）
    sw_orig_p <- NA_real_
    if(n_non_na >= 3 && n_non_na <= 5000) {
      try({
        sw_orig_p <- shapiro.test(x[!is.na(x)])$p.value
      }, silent = TRUE)
    }
    suggestions[i, "shapiro_p_orig"] <- sw_orig_p
    
    # 构造 log 版本（平移处理非正数）
    shifted <- 0
    if(!all(is.na(x))) {
      if(x_min <= 0) shifted <- -x_min + 1
      x_log <- log(x + shifted)
    } else {
      x_log <- x
    }
    
    # 画 log 后直方图
    if(save_plots) {
      hist(x_log, main = paste0(v, " (log shift=", shifted, ")"), xlab = paste0(v, "_log"), breaks = 30)
    }
    
    # Shapiro for log
    sw_log_p <- NA_real_
    if(n_non_na >= 3 && n_non_na <= 5000) {
      try({
        sw_log_p <- shapiro.test(x_log[!is.na(x_log)])$p.value
      }, silent = TRUE)
    }
    suggestions[i, "shapiro_p_log"] <- sw_log_p
    
    # 决策规则：
    # - 若样本量 >5000：用偏度阈值判断（skewness > skew_threshold）
    # - 若样本量 3-5000 且两者都有 shapiro p 值：若 log 的 shapiro p 值比原始高（改善超过 shapiro_improve_thresh），则推荐 log
    # - 或者若偏度本身很大（> skew_threshold），也推荐 log（即使 shapiro 没明显改善）
    recommend <- FALSE
    reason <- c()
    if(n_non_na > 5000) {
      if(!is.na(sk) && sk > skew_threshold) {
        recommend <- TRUE
        reason <- c(reason, paste0("样本量>", 5000, " 且偏度=", round(sk,3), " >", skew_threshold))
      } else {
        reason <- c(reason, paste0("样本量>", 5000, "，偏度=", round(sk,3)))
      }
    } else {
      # 有 shapiro p 值可比较
      if(!is.na(sw_orig_p) && !is.na(sw_log_p)) {
        if((sw_log_p - sw_orig_p) > shapiro_improve_thresh) {
          recommend <- TRUE
          reason <- c(reason, paste0("log 后 Shapiro p 从 ", round(sw_orig_p,4), " 提升到 ", round(sw_log_p,4)))
        } else {
          reason <- c(reason, paste0("log 后 Shapiro p 提升不显著（", round(sw_orig_p,4), "->", round(sw_log_p,4), "）"))
        }
      } else {
        # 若没有 shapiro p 值（样本量太小或其他），只看偏度
        if(!is.na(sk) && sk > skew_threshold) {
          recommend <- TRUE
          reason <- c(reason, paste0("样本量小/大于5000无法比较 Shapiro，但偏度=", round(sk,3), " >", skew_threshold))
        } else {
          reason <- c(reason, paste0("偏度=", round(sk,3)))
        }
      }
      # 无论如何，如果偏度非常大，作为补充判断也可推荐
      if(!recommend && !is.na(sk) && sk > (skew_threshold*1.5)) {
        recommend <- TRUE
        reason <- c(reason, paste0("偏度非常大，强烈推荐 log（skew=",round(sk,3),")"))
      }
    }
    
    suggestions[i, "recommend_log"] <- recommend
    suggestions[i, "reason"] <- paste(reason, collapse = "; ")
  } # end for
  
  # 关闭 pdf
  if(save_plots) dev.off()
  
  ## --------------------------
  ## 应用转换并标准化
  ## --------------------------
  # 先在一个副本中生成 log 变量并替换（如果推荐）
  data_trans <- data
  
  for(i in seq_len(nrow(suggestions))) {
    v <- suggestions$var[i]
    rec <- suggestions$recommend_log[i]
    if(isTRUE(rec)) {
      x <- data_trans[[v]]
      if(is.numeric(x)) {
        x_min <- min(x, na.rm = TRUE)
        shift <- ifelse(x_min <= 0, -x_min + 1, 0)
        data_trans[[v]] <- log(x + shift)
        # 记录替换说明（可在 suggestions 中添加列）
        suggestions$applied_shift[i] <- shift
      } else {
        suggestions$applied_shift[i] <- NA
      }
    } else {
      suggestions$applied_shift[i] <- NA
    }
  }
  
  # 标准化：对所有数值变量进行 scale（包括 response），保留分类变量原样
  is_num <- sapply(data_trans, is.numeric)
  # 若用户指定某些列为分类变量，但实际上是数值，则排除
  if(length(categorical_vars) > 0) {
    is_num[categorical_vars] <- FALSE
  }
  num_names <- names(is_num)[is_num]
  cat("以下数值列将被标准化：\n"); print(num_names)
  
  data_scaled <- data_trans
  if(length(num_names) > 0) {
    scaled_mat <- scale(data_trans[, num_names, drop = FALSE])
    # scale 可能包含 NA 的列（全 NA 或 sd=0），用原值替代
    for(j in seq_along(num_names)) {
      cname <- num_names[j]
      col_scaled <- scaled_mat[, j]
      if(all(is.na(col_scaled))) {
        # 无法标准化，保留原始
        warning(paste0("列 ", cname, " 无法标准化（可能全 NA 或常数），保留原值。"))
        data_scaled[[cname]] <- data_trans[[cname]]
      } else {
        data_scaled[[cname]] <- as.numeric(col_scaled)
      }
    }
  }
  
  # 保存建议表与最终表
  write.csv(suggestions, file = suggestion_csv, row.names = FALSE)
  write.csv(data_scaled, file = out_csv, row.names = FALSE)
  
  cat("\n处理完成。\n")
  cat("建议表已保存为：", suggestion_csv, "\n")
  cat("最终 log(已替换)+标准化 数据表已保存为：", out_csv, "\n")
  if(save_plots) cat("诊断图已保存为：", plot_pdf, "\n")
  
  return(ions,
    data_transformed = data_trans,
    data_scaled = data_scaled
  ))
}
